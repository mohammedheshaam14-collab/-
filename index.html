        function calcBuy() {
            let grand = 0;
            for(let i=1; i<=5; i++) {
                const w = parseFloat(document.getElementById(`cb_weight_${i}`).value) || 0;
                const karat = document.getElementById(`cb_karat_${i}`).value;
                const cp = parseFloat(document.getElementById(`cb_price_${i}`).value) || state.k21Price;
                const work = parseFloat(document.getElementById(`cb_work_${i}`).value) || 0;
                let gp = karat === '24' ? (cp*24)/21 : (karat === '18' ? (cp*6)/7 : cp);
                const tot = calculateItemPrice(w, gp, work);
                grand += tot;
                document.getElementById(`cb_res_${i}`).innerText = Math.round(tot).toLocaleString('ar-EG') + ' ج.م';
            }
            document.getElementById('calcBuyGrandTotal').innerText = Math.round(grand).toLocaleString('ar-EG') + ' ج.م';
        }

        function calcSwap() {
            let oldTot = 0, newTot = 0;
            for(let i=1; i<=3; i++) {
                const ow = parseFloat(document.getElementById(`so_w_${i}`).value) || 0;
                const op = parseFloat(document.getElementById(`so_p_${i}`).value) || 0;
                const od = parseFloat(document.getElementById(`so_d_${i}`).value) || 0;
                oldTot += calculateItemPrice(ow, Math.max(0, op - od), 0);
                const nw = parseFloat(document.getElementById(`sn_w_${i}`).value) || 0;
                const np = parseFloat(document.getElementById(`sn_p_${i}`).value) || 0;
                const nk = parseFloat(document.getElementById(`sn_k_${i}`).value) || 0;
                newTot += calculateItemPrice(nw, np, nk);
            }
            document.getElementById('swapOldTotalRes').innerText = Math.round(oldTot).toLocaleString('ar-EG') + ' ج.م';
            document.getElementById('swapNewTotalRes').innerText = Math.round(newTot).toLocaleString('ar-EG') + ' ج.م';
            const diff = newTot - oldTot;
            const lbl = document.getElementById('swapFinalStatusLabel');
            const res = document.getElementById('swapFinalResult');
            if(diff > 0) {
                lbl.innerText = "الزبون سيقوم بدفع (فارق نقدي):"; res.className = "text-2xl font-black text-rose-400"; res.innerText = Math.round(diff).toLocaleString('ar-EG') + ' ج.م (دفع)';
            } else {
                lbl.innerText = "الزبون سيأخذ (فارق نقدي):"; res.className = "text-2xl font-black text-emerald-400"; res.innerText = Math.round(Math.abs(diff)).toLocaleString('ar-EG') + ' ج.م (قبض)';
            }
        }

        function handleBuyGold(e) {
            e.preventDefault();
            const karat = document.getElementById('buyKarat').value;
            const weight = parseFloat(document.getElementById('buyWeight').value) || 0;
            const itemName = document.getElementById('buyManualItemName').value.trim() || 'قطعة';
            const gPrice = karat === '24' ? (state.k21Price * 24)/21 : (karat === '18' ? (state.k21Price * 6)/7 : state.k21Price);
            let cost = calculateItemPrice(weight, gPrice, 0);
            if (manualTotalChanged) cost = parseFloat(document.getElementById('buyCustomTotalOverride').value) || cost;
            if (cost > state.vaultCash) return alert('عفواً، لا توجد نقدية كافية بالخزينة!');

            const imageInput = document.getElementById('buyItemImage');
            if(imageInput.files && imageInput.files[0]) {
                const reader = new FileReader();
                reader.onload = function(uploadEvent) {
                    saveItemToInventory(itemName, karat, weight, cost, uploadEvent.target.result);
                };
                reader.readAsDataURL(imageInput.files[0]);
            } else {
                saveItemToInventory(itemName, karat, weight, cost, null);
            }
        }

            e.preventDefault();
            const selectedId = document.getElementById('sellItemSelect').value;

        function updateItemName(id, val) {
            const item = state.inventory.find(x => x.id === id);
            if(item) { item.itemName = val; saveState(); populateAiStudioDropdown(); }
        }
        function updateItemWeight(id, val) {
            const item = state.inventory.find(x => x.id === id);
            if(item) { item.weight = parseFloat(val) || item.weight; item.pureGoldWeight = (item.weight * parseInt(item.karat)) / 24; saveState(); renderDashboard(); }
        }
        function updateItemCost(id, val) {
            const item = state.inventory.find(x => x.id === id);
            if(item) { item.purchaseCost = parseFloat(val) || item.purchaseCost; saveState(); }
        }

        function deleteInventoryItem(id) {
            if(confirm('هل أنت متأكد من حذف هذه القطعة من المخزون؟')) {
                state.inventory = state.inventory.filter(x => x.id !== id);
                saveState(); renderAllData(); populateAiStudioDropdown();
            }
        }

        function renderVaultTransactions() {
            const tbody = document.getElementById('vaultTransactionsBody'); tbody.innerHTML = '';
            state.transactions.forEach((t, index) => {
                const col = t.amount > 0 ? 'text-emerald-400' : 'text-rose-400';
                tbody.innerHTML += `
                    <tr class="hover:bg-dark-700/50 transition">
                        <td class="p-3 text-gray-400">${t.date}</td>
                        <td class="p-3">${t.type}</td>
                        <td class="p-3"><input type="text" value="${t.desc}" onchange="updateTxDesc(${index}, this.value)" class="bg-dark-900 border border-gray-700 text-white px-2 py-1 rounded w-full text-xs focus:outline-none"></td>
                        <td class="p-3 font-black ${col}"><input type="number" value="${Math.round(t.amount)}" onchange="updateTxAmount(${index}, this.value)" class="bg-dark-900 border border-gray-700 text-white px-2 py-1 rounded w-28 text-xs font-bold focus:outline-none"> ج.م</td>
                        <td class="p-3 text-center">
                            <button onclick="deleteTx(${index})" class="text-rose-400 hover:text-rose-300 bg-rose-950/40 px-2.5 py-1 rounded-lg border border-rose-800/50"><i class="fa-solid fa-trash"></i></button>
                        </td>
                    </tr>
                `;
            });
        }

        function updateTxDesc(index, val) { state.transactions[index].desc = val; saveState(); }
        function updateTxAmount(index, val) { 
            const diff = (parseFloat(val) || 0) - state.transactions[index].amount;
            state.transactions[index].amount = parseFloat(val) || 0;
            state.vaultCash += diff;
            saveState(); renderDashboard(); 
        }
        function deleteTx(index) {
            if(confirm('حذف هذه الحركة النقدية؟')) {
                const amt = state.transactions[index].amount;
                state.vaultCash -= amt;
                state.transactions.splice(index, 1);
                saveState(); renderAllData();
            }
        }
